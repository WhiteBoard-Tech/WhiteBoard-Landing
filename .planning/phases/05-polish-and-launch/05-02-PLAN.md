<task type="auto">
  <name>Task 1: Configure Vercel adapter and middleware</name>
  <files>astro.config.mjs, src/middleware.ts, vercel.json, package.json</files>
  <action>
    1. Add Vercel adapter for edge middleware support:
       `npx astro add vercel`
       - Select "yes" to install the adapter
       - This will update astro.config.mjs automatically

    2. Verify/update astro.config.mjs:
       ```javascript
       import { defineConfig } from "astro/config";
       import tailwindcss from "@tailwindcss/vite";
       import vercel from "@astrojs/vercel";

       export default defineConfig({
         output: "static",
         adapter: vercel({
           edgeMiddleware: true,
         }),
         i18n: {
           locales: ["en", "es"],
           defaultLocale: "en",
           routing: {
             prefixDefaultLocale: true,
             redirectToDefaultLocale: false, // We handle redirect in middleware
           },
         },
         vite: {
           plugins: [tailwindcss()],
         },
       });
       ```

    3. Create src/middleware.ts for Accept-Language detection:
       ```typescript
       import { defineMiddleware } from "astro:middleware";

       export const onRequest = defineMiddleware((context, next) => {
         const url = new URL(context.request.url);

         // Only redirect root path
         if (url.pathname === "/") {
           const acceptLanguage = context.request.headers.get("accept-language") || "";
           // Check if Spanish is preferred (includes es, es-ES, es-MX, etc.)
           const preferredLocale = acceptLanguage.toLowerCase().includes("es") ? "es" : "en";
           return context.redirect(`/${preferredLocale}/`, 302);
         }

         return next();
       });
       ```

    4. Create vercel.json for caching headers:
       ```json
       {
         "headers": [
           {
             "source": "/(.*)\\.(?:js|css|woff2|avif|webp|svg|png|jpg|jpeg)$",
             "headers": [
               {
                 "key": "Cache-Control",
                 "value": "public, max-age=31536000, immutable"
               }
             ]
           }
         ]
       }
       ```

    5. Test locally:
       - `npm run build`
       - `npm run preview`
       - Verify http://localhost:4321/ redirects to /en/ (default for most browsers)
  </action>
