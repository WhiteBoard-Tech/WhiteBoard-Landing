2. Create `src/i18n/utils.ts`:

```typescript
import { ui, defaultLang, type Lang, type TranslationKey } from "./ui";

/**
 * Extract language code from URL pathname
 * /en/about -> "en"
 * /es/services -> "es"
 */
export function getLangFromUrl(url: URL): Lang {
  const [, lang] = url.pathname.split("/");
  if (lang && lang in ui) {
    return lang as Lang;
  }
  return defaultLang;
}

/**
 * Get translation function for a specific language
 * Usage: const t = useTranslations(lang); t("site.title")
 */
export function useTranslations(lang: Lang) {
  return function t(key: TranslationKey): string {
    return ui[lang][key] || ui[defaultLang][key];
  };
}

/**
 * Get the pathname without the locale prefix
 * /en/about -> /about
 * /es/ -> /
 */
export function getPathWithoutLocale(pathname: string): string {
  const segments = pathname.split("/").filter(Boolean);
  if (segments[0] && segments[0] in ui) {
    segments.shift();
  }
  return "/" + segments.join("/") || "/";
}
```

The type system ensures:
- All translation keys are type-checked
- Missing translations fall back to English
- Language codes are validated

2. Create `src/components/LanguageSwitcher.astro`:

```astro
---
import { languages, defaultLang } from "@i18n/ui";
import { getPathWithoutLocale } from "@i18n/utils";

const currentLang = Astro.currentLocale || defaultLang;
const currentPath = Astro.url.pathname;
const basePath = getPathWithoutLocale(currentPath);
---

<nav class="flex items-center gap-1" aria-label="Language selector">
  {Object.entries(languages).map(([code, name]) => (
    <a
      href={`/${code}${basePath === "/" ? "/" : basePath}`}
      class:list={[
        "px-3 py-1.5 text-sm font-medium rounded transition-colors duration-200",
        currentLang === code
          ? "bg-black text-white"
          : "text-gray-600 hover:text-black hover:bg-gray-100",
      ]}
      aria-current={currentLang === code ? "page" : undefined}
      hreflang={code}
    >
      {code.toUpperCase()}
    </a>
  ))}
</nav>
```

Key implementation details:
- Mobile-first responsive classes (px-4 sm:px-6 lg:px-8)
- Fixed header with backdrop blur for premium feel
- Accessible skip link for keyboard navigation
- Language switcher uses getPathWithoutLocale to avoid double-prefixing
- aria-current for screen reader support
