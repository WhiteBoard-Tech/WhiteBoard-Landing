Create `src/scripts/calendly.ts` with facade pattern for lazy-loading:

```typescript
let calendlyLoaded = false;
let calendlyLoading: Promise<void> | null = null;

export async function loadCalendly(): Promise<void> {
  if (calendlyLoaded) return;
  if (calendlyLoading) return calendlyLoading;

  calendlyLoading = new Promise((resolve, reject) => {
    // Load CSS first
    const link = document.createElement('link');
    link.rel = 'stylesheet';
    link.href = 'https://assets.calendly.com/assets/external/widget.css';
    document.head.appendChild(link);

    // Load JS
    const script = document.createElement('script');
    script.src = 'https://assets.calendly.com/assets/external/widget.js';
    script.async = true;
    script.onload = () => {
      calendlyLoaded = true;
      resolve();
    };
    script.onerror = () => reject(new Error('Failed to load Calendly widget'));
    document.head.appendChild(script);
  });

  return calendlyLoading;
}

export function openCalendlyPopup(locale: string): void {
  const baseUrl = 'https://calendly.com/whiteboard-co/30min';
  const url = `${baseUrl}?locale=${locale}`;
  window.Calendly.initPopupWidget({ url });
}
```

Add Calendly type declaration to `src/env.d.ts` (append to existing file):

```typescript
declare global {
  interface Window {
    Calendly: {
      initPopupWidget: (options: { url: string }) => void;
      closePopupWidget: () => void;
    };
  }
}
```

  type="button"
  id="calendly-cta"
  data-calendly-locale={lang}
  class="inline-flex items-center justify-center px-8 py-4 text-base font-semibold text-black bg-white rounded hover:bg-gray-100 transition-colors duration-200 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-offset-black focus:ring-white disabled:opacity-50 disabled:cursor-wait"

  import { loadCalendly, openCalendlyPopup } from '@scripts/calendly';

  const button = document.getElementById('calendly-cta') as HTMLButtonElement;
  const locale = button.dataset.calendlyLocale || 'en';

  button.addEventListener('click', async () => {
    button.disabled = true;
    try {
      await loadCalendly();
      openCalendlyPopup(locale);
    } catch (error) {
      console.error('Failed to load Calendly:', error);
    } finally {
      button.disabled = false;
    }
  });
</script>
